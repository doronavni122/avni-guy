C3: python3 - <<'PY'
import re,json,sys
html=open('/tmp/avniguy-about.html',encoding='utf-8').read()
blocks=re.findall(r'<script[^>]*type="application/ld\+json"[^>]*>(.*?)</script>',html,re.S|re.I)
people=[]
for b in blocks:
  try: data=json.loads(b)
  except Exception as e: print('JSON_FAIL',e); continue
  stack=data if isinstance(data,list) else [data]
  for node in stack:
    if isinstance(node,dict) and node.get('@graph'):
      stack.extend(node['@graph'])
    t=node.get('@type') if isinstance(node,dict) else None
    types=t if isinstance(t,list) else ([t] if t else [])
    if 'Person' in types: people.append(node)
print('PERSON_COUNT',len(people))
for p in people:
  print('SAMEAS',json.dumps(p.get('sameAs'),ensure_ascii=False))
  print('URL',p.get('url')); print('ID',p.get('@id'))
if not people: sys.exit(2)
PY
